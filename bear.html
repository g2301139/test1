// === 7. 雨雲レーダー予測の取得と生成 (15時間先対応・修正版) === async function fetchJmaRadarTimestamps() { const ts = Date.now(); // N1: 直近（ナウキャスト）、N2: 1〜6時間先、N3: 7〜15時間先 // どのURLから取得したかを識別するため、defaultTypeを手動で付与します const fetchList = [ { url: `https://www.jma.go.jp/bosai/jmatile/data/nowc/targetTimes_N1.json?_=${ts}`, defaultType: 'hrpns' }, { url: `https://www.jma.go.jp/bosai/jmatile/data/nowc/targetTimes_N2.json?_=${ts}`, defaultType: 'hrpns_nd' }, { url: `https://www.jma.go.jp/bosai/jmatile/data/nowc/targetTimes_N3.json?_=${ts}`, defaultType: 'hrpns_nd' } ]; let allTargets = []; for (const item of fetchList) { try { const res = await fetch(item.url); if (res.ok) { const data = await res.json(); if (Array.isArray(data)) { // 取得したデータにデフォルトの要素名を埋め込む（気象庁側で省略されていても対応するため） data.forEach(d => { d._fallbackType = item.defaultType; }); allTargets = allTargets.concat(data); } } } catch (e) { console.warn(`Failed to fetch radar data: ${item.url}`, e); } } const syncInfo = document.getElementById('slider-sync-info'); if (allTargets.length === 0) { syncInfo.textContent = "レーダー情報を取得できませんでした"; syncInfo.style.color = "#dc2626"; syncInfo.style.borderColor = "#f87171"; syncInfo.style.backgroundColor = "#fef2f2"; showErrorToast("雨雲レーダーのデータ取得に失敗しました"); return; } // 1. 並び替え：未来に向かって時間順（古い順） // 2. 同じ有効時間なら、basetime（発表時間）が「新しい方」を優先 allTargets.sort((a, b) => { if (a.validtime === b.validtime) { return a.basetime < b.basetime ? 1 : -1; // basetimeは降順 } return a.validtime < b.validtime ? -1 : 1; // validtimeは昇順 }); // validtime（予報時間）で重複を排除。新しい発表時間のものが残る const uniqueTargets = []; const seen = new Set(); allTargets.forEach(t => { if (!seen.has(t.validtime)) { seen.add(t.validtime); uniqueTargets.push(t); } }); const now = new Date(); js_urls = []; js_labels = []; js_dates = []; uniqueTargets.forEach(t => { // UTCからローカル時間（JST）へ変換 const vDate = new Date(Date.UTC( parseInt(t.validtime.substring(0,4)), parseInt(t.validtime.substring(4,6)) - 1, parseInt(t.validtime.substring(6,8)), parseInt(t.validtime.substring(8,10)), parseInt(t.validtime.substring(10,12)) )); const diffMins = Math.round((vDate - now) / 60000); if (diffMins < -60) return; // 1時間以上前の過去データは不要 // JSON内に要素名があれば優先、なければURLに基づくfallbackを使用 let element = t._fallbackType; if (t.elements && Array.isArray(t.elements)) { if (t.elements.includes("hrpns_nd")) element = "hrpns_nd"; else if (t.elements.includes("hrpns")) element = "hrpns"; } js_urls.push(`https://www.jma.go.jp/bosai/jmatile/data/nowc/${t.basetime}/none/${t.validtime}/surf/${element}/{z}/{x}/{y}.png`); js_dates.push(vDate); js_labels.push(formatTimeLabel(vDate, diffMins)); }); const slider = document.getElementById('radar-time-slider'); slider.max = js_urls.length - 1; let closestIdx = 0, minDiff = Infinity; js_dates.forEach((d, i) => { const diff = Math.abs(d - now); if (diff < minDiff) { minDiff = diff; closestIdx = i; } }); slider.value = closestIdx; document.getElementById('time-label').textContent = js_labels[closestIdx]; syncInfo.textContent = `過去1時間 〜 最大15時間先までの予測 (${js_urls.length}コマ)`; }
