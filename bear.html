// ========================================== // 2. 雨雲データの時間を取得する関数（15時間先まで取得） // ========================================== async function fetchJmaRadarTimestamps() { try { const ts = Date.now(); // nowc: 降水ナウキャスト（5分刻み・1時間先まで） // prca: 降水短時間予報（1時間刻み・1〜15時間先まで） const targetApis = [ { prod: "nowc", element: "hrpns", url: `https://www.jma.go.jp/bosai/jmatile/data/nowc/targetTimes_N1.json?_=${ts}` }, { prod: "nowc", element: "hrpns", url: `https://www.jma.go.jp/bosai/jmatile/data/nowc/targetTimes_N2.json?_=${ts}` }, { prod: "nowc", element: "hrpns", url: `https://www.jma.go.jp/bosai/jmatile/data/nowc/targetTimes_N3.json?_=${ts}` }, { prod: "prca", element: "prca", url: `https://www.jma.go.jp/bosai/jmatile/data/prca/targetTimes_N1.json?_=${ts}` }, { prod: "prca", element: "prca", url: `https://www.jma.go.jp/bosai/jmatile/data/prca/targetTimes_N2.json?_=${ts}` }, { prod: "prca", element: "prca", url: `https://www.jma.go.jp/bosai/jmatile/data/prca/targetTimes_N3.json?_=${ts}` }, { prod: "prca", element: "prca", url: `https://www.jma.go.jp/bosai/jmatile/data/prca/targetTimes.json?_=${ts}` } ]; const responses = await Promise.all(targetApis.map(t => fetch(t.url).catch(() => null))); let allTimes = {}; for (let i = 0; i < responses.length; i++) { const res = responses[i]; if (!res || !res.ok) continue; const data = await res.json(); if (!Array.isArray(data)) continue; const prod = targetApis[i].prod; const defaultElement = targetApis[i].element; data.forEach(t => { if (t.basetime && t.validtime) { // 詳細なnowcデータ（5分刻み）を優先し、prca（1時間刻み）で上書きしないようにする if (!allTimes[t.validtime] || prod === "nowc") { t.prod = prod; t.element = t.elements ? t.elements[0] : defaultElement; allTimes[t.validtime] = t; } } }); } // 時間順に並び替え let sortedTimes = Object.values(allTimes).sort((a, b) => a.validtime.localeCompare(b.validtime)); const now = new Date(); js_urls = []; js_labels = []; sortedTimes.forEach((t) => { const year = parseInt(t.validtime.substring(0,4)); const month = parseInt(t.validtime.substring(4,6)) - 1; const day = parseInt(t.validtime.substring(6,8)); const hour = parseInt(t.validtime.substring(8,10)); const min = parseInt(t.validtime.substring(10,12)); const validDateUtc = new Date(Date.UTC(year, month, day, hour, min)); const validDateJst = new Date(validDateUtc.getTime()); // 現在時刻との差分（分）を計算 const diffMins = Math.round((validDateJst - now) / 60000); // 過去15分以上前の古いデータはスライダーから除外 if (diffMins < -15) return; const timeStr = `${String(validDateJst.getHours()).padStart(2,'0')}:${String(validDateJst.getMinutes()).padStart(2,'0')}`; js_urls.push(`https://www.jma.go.jp/bosai/jmatile/data/${t.prod}/${t.basetime}/none/${t.validtime}/surf/${t.element}/{z}/{x}/{y}.png`); if (diffMins <= 5 && diffMins >= -15) { js_labels.push(`現在 (${timeStr})`); } else if (diffMins > 5 && diffMins < 60) { js_labels.push(`${diffMins}分後 (${timeStr})`); } else { const hDiff = Math.floor(diffMins / 60); const mDiff = diffMins % 60; if (mDiff === 0) js_labels.push(`${hDiff}時間後 (${timeStr})`); else js_labels.push(`${hDiff}時間${mDiff}分後 (${timeStr})`); } }); // スライダーの初期設定 const slider = document.getElementById('radar-time-slider'); if (slider && js_urls.length > 0) { slider.max = js_urls.length - 1; // 「現在」のインデックスを自動で探して初期値に設定（見つからなければ0） const currentIndex = js_labels.findIndex(label => label.includes("現在")); slider.value = currentIndex !== -1 ? currentIndex : 0; const timeLabel = document.getElementById('time-label'); if (timeLabel) timeLabel.textContent = js_labels[slider.value]; if (isRadarActive) updateRadarLayer(); } } catch (e) { console.error("雨雲データの取得に失敗しました。", e); } } // ========================================== // 3. 雨雲レーダーを地図に表示・更新する関数 // ========================================== function updateRadarLayer() { const slider = document.getElementById('radar-time-slider'); if (radarLayer && typeof map !== 'undefined' && map.hasLayer(radarLayer)) { map.removeLayer(radarLayer); } const url = js_urls[slider.value]; if (!url) return; // URLに「prca（15時間予報）」が含まれているか判定 const isPrca = url.includes('/prca/'); // 1時間先以降の予報(prca)は詳細なズーム画像が存在しないため、上限を6に制限して自動引き伸ばしさせる const maxZ = isPrca ? 6 : 10; if (typeof map !== 'undefined') { radarLayer = L.tileLayer(url, { opacity: 0.65, zIndex: 1500, maxNativeZoom: maxZ }).addTo(map); } }
